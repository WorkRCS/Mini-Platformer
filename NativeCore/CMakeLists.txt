cmake_minimum_required(VERSION 3.20)
project(WildboundCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_library(WildboundCore SHARED src/WildboundCore.cpp)
target_include_directories(WildboundCore PUBLIC "include")
target_compile_definitions(WildboundCore PRIVATE WILDBOUND_CORE_EXPORTS)
set_target_properties(WildboundCore PROPERTIES OUTPUT_NAME "WildboundCore" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
include(CTest)
if(BUILD_TESTING)
 add_executable(WildboundCoreTests tests/WildboundCoreTests.cpp src/WildboundCore.cpp)
 target_include_directories(WildboundCoreTests PRIVATE "include")
 target_compile_definitions(WildboundCoreTests PRIVATE WILDBOUND_CORE_EXPORTS)
 add_test(NAME WildboundCoreTests COMMAND WildboundCoreTests)
endif()
